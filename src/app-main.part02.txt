ers() {
  const query = state.query.normalize('NFKC').toLocaleLowerCase('ja').trim();
  state.filtered = state.catalog.works.filter((work) => {
    const haystack = [work.title, work.author ?? '', work.category, ...work.sources].join(' ').normalize('NFKC').toLocaleLowerCase('ja');
    return (!query || haystack.includes(query))
      && (!state.statuses.size || state.statuses.has(work.status))
      && (!state.categories.size || state.categories.has(work.category))
      && (!state.sources.size || work.sources.some((source) => state.sources.has(source)));
  });
  const rank = { read: 4, reading: 3, unread: 2, untracked: 1 };
  state.filtered.sort((a, b) => {
    if (state.sort === 'title-asc') return a.title.localeCompare(b.title, 'ja');
    if (state.sort === 'rating-desc') return (b.rating ?? -1) - (a.rating ?? -1) || a.title.localeCompare(b.title, 'ja');
    if (state.sort === 'price-desc') return (b.price_yen ?? -1) - (a.price_yen ?? -1) || a.title.localeCompare(b.title, 'ja');
    if (state.sort === 'status-desc') return rank[b.status] - rank[a.status] || a.title.localeCompare(b.title, 'ja');
    if (state.sort === 'items-desc') return b.item_count - a.item_count || a.title.localeCompare(b.title, 'ja');
    return (b.acquired_at ?? '').localeCompare(a.acquired_at ?? '') || a.title.localeCompare(b.title, 'ja');
  });
  const maxPage = Math.max(1, Math.ceil(state.filtered.length / PAGE_SIZE));
  state.page = Math.min(state.page, maxPage);
  renderBooks(); renderActiveFilters(); renderPagination(); syncUrl();
}

function coverColor(work) {
  let hash = 0;
  for (const char of work.title_key) hash = ((hash << 5) - hash + char.charCodeAt(0)) | 0;
  return palettes[Math.abs(hash) % palettes.length];
}

function bookCard(work, index) {
  const card = document.createElement('button');
  card.type = 'button'; card.className = 'book-card';
  card.setAttribute('aria-label', `${work.title}の詳細を見る`);
  card.innerHTML = `
    <div class="book-cover" style="--cover:${coverColor(work)}">
      <div class="cover-inner"><span class="cover-category">${escapeHtml(work.category)}</span><strong class="cover-title">${escapeHtml(work.title)}</strong><span class="cover-index">KAFKA / ${String(index + 1).padStart(3,'0')}</span></div>
    </div>
    <div class="book-meta"><h3>${escapeHtml(work.title)}</h3><div class="meta-row"><span class="status-dot" style="--status-color:${statusColors[work.status]}">${statusLabels[work.status]}</span>${work.item_count > 1 ? `<span class="count-badge">${work.item_count}件統合</span>` : '<span></span>'}</div></div>`;
  card.addEventListener('click', () => showDetail(work));
  return card;
}

function renderBooks() {
  const grid = $('#bookGrid'); grid.innerHTML = '';
  grid.classList.toggle('is-list', state.view === 'list');
  const start = (state.page - 1) * PAGE_SIZE;
  state.filtered.slice(start, start + PAGE_SIZE).forEach((work, index) => grid.append(bookCard(work, start + index)));
  $('#resultCount').textContent = fmtNumber.format(state.filtered.length);
  $('#emptyState').hidden = state.filtered.length !== 0;
}

function renderActiveFilters() {
  const values = [];
  if (state.query) values.push(`検索: ${state.query}`);
  [...state.statuses].forEach((value) => values.push(statusLabels[value]));
  [...state.categories].forEach((value) => values.push(value));
  [...state.sources].forEach((value) => values.push(value));
  $('#activeFilters').innerHTML = values.map((value) => `<span class="filter-chip">${escapeHtml(value)}</span>`).join('');
}

function renderPagination() {
  const total = Math.ceil(state.filtered.length / PAGE_SIZE);
  const nav = $('#pagination'); nav.innerHTML = '';
  if (total <= 1) return;
  const pages = new Set([1, total, state.page - 1, state.page, state.page + 1]);
  for (const page of [...pages].filter((value) => value >= 1 && value <= total).sort((a,b) => a-b)) {
    const button = document.createElement('button'); button.className = 'page-button'; button.type = 'button'; button.textContent = page;
    if (page === state.page) button.setAttribute('aria-current', 'page');
    button.addEventListener('click', () => { state.page = page; applyFilters(); $('#catalog').scrollIntoView(); });
    nav.append(button);
  }
}
