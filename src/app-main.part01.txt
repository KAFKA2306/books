async function loadCatalog() {
  const manifest = await fetch('./data/catalog.parts.json').then((response) => response.json());
  const chunks = await Promise.all(manifest.parts.map((part) => fetch(`./data/${part}`).then((response) => response.text())));
  const binary = atob(chunks.join('').replace(/\s/g, ''));
  const bytes = Uint8Array.from(binary, (character) => character.charCodeAt(0));
  const stream = new Blob([bytes]).stream().pipeThrough(new DecompressionStream('gzip'));
  const compact = await new Response(stream).json();
  const works = compact.w.map(([work_id,title,title_key,category,status,progress,rating,price_yen,acquired_at,item_count,sources,formats]) => ({ work_id,title,title_key,author:null,category,status,progress,rating,price_yen,acquired_at,item_count,isbn_count:0,isbn_status:'missing',sources,formats }));
  const editions = works.map((work) => ({ edition_id:`pending:${work.work_id.slice(4)}`,id_kind:'pending_title_key',work_id:work.work_id,isbn13:null,verification:'unverified' }));
  const holdings = works.flatMap((work) => work.sources.map((source, index) => ({ holding_id:`${work.work_id}:h${index+1}`,work_id:work.work_id,edition_id:`pending:${work.work_id.slice(4)}`,source,format:work.formats[index] ?? work.formats[0] ?? '不明',quantity:index===0?work.item_count:1,acquired_at:work.acquired_at,price_yen:index===0?work.price_yen:null,progress:index===0?work.progress:null,rating:index===0?work.rating:null })));
  const issueManifest = await fetch('./data/issue-1-books.parts.json').then((response) => response.json());
  const issueEncoded = (await Promise.all(issueManifest.parts.map((part) => fetch(`./data/${part}`).then((response) => response.text())))).join('');
  const issueBinary = atob(issueEncoded.replace(/\s/g, ''));
  const issueBytes = Uint8Array.from(issueBinary, (character) => character.charCodeAt(0));
  const issueStream = new Blob([issueBytes]).stream().pipeThrough(new DecompressionStream('gzip'));
  const issueData = await new Response(issueStream).json();
  return mergeIssueCatalog({ schema_version:compact.v,generated_at:compact.g,stats:{...compact.s,edition_count:editions.length,holding_count:holdings.length},works,editions,holdings }, issueData);
}

function hydrateFromUrl() {
  const params = new URLSearchParams(location.search);
  state.query = params.get('q') ?? '';
  state.statuses = new Set(params.getAll('status'));
  state.categories = new Set(params.getAll('category'));
  state.sources = new Set(params.getAll('source'));
  state.sort = params.get('sort') ?? 'acquired-desc';
  state.view = params.get('view') === 'list' ? 'list' : 'grid';
  state.page = Math.max(1, Number(params.get('page') ?? 1));
}

function syncUrl() {
  const params = new URLSearchParams();
  if (state.query) params.set('q', state.query);
  [...state.statuses].forEach((value) => params.append('status', value));
  [...state.categories].forEach((value) => params.append('category', value));
  [...state.sources].forEach((value) => params.append('source', value));
  if (state.sort !== 'acquired-desc') params.set('sort', state.sort);
  if (state.view !== 'grid') params.set('view', state.view);
  if (state.page > 1) params.set('page', state.page);
  history.replaceState(null, '', `${location.pathname}${params.size ? `?${params}` : ''}`);
}

function countsFor(field) {
  const counts = new Map();
  for (const work of state.catalog.works) {
    const values = Array.isArray(work[field]) ? work[field] : [work[field]];
    for (const value of values) if (value) counts.set(value, (counts.get(value) ?? 0) + 1);
  }
  return counts;
}

function createChecks(container, entries, selected, labeler = (value) => value) {
  container.innerHTML = '';
  for (const [value, count] of entries) {
    const row = document.createElement('label');
    row.className = 'check-row';
    row.innerHTML = `<input type="checkbox" value="${escapeHtml(value)}" ${selected.has(value) ? 'checked' : ''}><span>${escapeHtml(labeler(value))}</span><em>${fmtNumber.format(count)}</em>`;
    row.querySelector('input').addEventListener('change', (event) => {
      event.target.checked ? selected.add(value) : selected.delete(value);
      state.page = 1;
      applyFilters();
    });
    container.append(row);
  }
}

function setupFilters() {
  createChecks($('#statusFilters'), Object.entries(countsFor('status')), state.statuses, (value) => statusLabels[value]);
  createChecks($('#categoryFilters'), [...countsFor('category')].sort((a,b) => b[1] - a[1] || a[0].localeCompare(b[0], 'ja')), state.categories);
  createChecks($('#sourceFilters'), [...countsFor('sources')].sort((a,b) => b[1] - a[1]), state.sources);
  $('#searchInput').value = state.query;
  $('#sortSelect').value = state.sort;
  setView(state.view);
}
